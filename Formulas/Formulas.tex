\section{TLine}

\subsection*{Transmission Line Parameters}
\begin{itemize}
    \item \textbf{Resistance per unit length:} $R_0 \; [\Omega/\text{m}]$
    \item \textbf{Inductance per unit length:} $L_0 \; [\text{H}/\text{m}]$
    \item \textbf{Capacitance per unit length:} $C_0 \; [\text{F}/\text{m}]$
    \item \textbf{Conductance per unit length:} $G_0 \; [\text{S}/\text{m}]$
\end{itemize}

\subsection*{The Telegrapher Equations}
\begin{itemize}
    \item \textbf{Voltage equation:} 
    \[
    \frac{\partial V(z,t)}{\partial z} = -R_0\, i(z,t) - L_0 \frac{\partial i(z,t)}{\partial t}
    \]
    \item \textbf{Current equation:}
    \[
    \frac{\partial i(z,t)}{\partial z} = -G_0\, V(z,t) - C_0 \frac{\partial V(z,t)}{\partial t}
    \]
    \item \textbf{Lossless line:} for $R_0 = 0$ and $G_0 = 0$,
    \[
    \frac{\partial^2 V}{\partial z^2} = L_0 C_0 \frac{\partial^2 V}{\partial t^2}
    \]
\end{itemize}

\subsection*{Wave Propagation Dynamics}
\begin{itemize}
    \item \textbf{Propagation speed:}
    \[
    v = \frac{1}{\sqrt{L_0 C_0}}
    \]
    \item \textbf{Voltage solution:}
    \[
    V(z,t) = V^+(t - z/v) + V^-(t + z/v)
    \]
    \item \textbf{Characteristic impedance:}
    \[
    Z_0 = \sqrt{\frac{L_0}{C_0}}
    \]
    \item \textbf{Current waves:}
    \[
    i^+(z,t) = \frac{V^+(z,t)}{Z_0}, \qquad
    i^-(z,t) = -\frac{V^-(z,t)}{Z_0}
    \]
\end{itemize}

\subsection*{Reflections and Terminations}
\begin{itemize}
    \item \textbf{Voltage reflection coefficient:}
    \[
    \rho_v = \Gamma_L = \frac{R_L - Z_0}{R_L + Z_0}
    \]
    \item \textbf{Current reflection coefficient:}
    \[
    \rho_i = -\rho_v = \frac{Z_0 - R_L}{Z_0 + R_L}
    \]
    \item \textbf{Matched load:} if $R_L = Z_0$, then $\Gamma_L = 0$ and no reflections occur.
\end{itemize}

\subsection*{Power on the Transmission Line}
\begin{itemize}
    \item \textbf{Instantaneous power:} 
    \[
    p(z,t) = V(z,t)\, i(z,t)
    \]
    \item \textbf{Average power for a traveling wave:}
    \[
    P_{\text{avg}} = \frac{V_{\mathrm{rms}}^2}{Z_0} = I_{\mathrm{rms}}^2 Z_0
    \]
    \item \textbf{Power transfer:} maximum power is delivered when the line is matched to the load.
\end{itemize}

\section{Smatrix}
\subsection*{Scattering Parameters for an Impedance Discontinuity}
\begin{itemize}
    \item \textbf{Impedances:} Two semi-infinite lines meet at a junction with characteristic impedances $Z_1$ and $Z_2$.
    \item \textbf{Power-normalized waves:}
    \[
    a_n = \frac{V_n^+}{\sqrt{Z_n}}, \qquad b_n = \frac{V_n^-}{\sqrt{Z_n}}
    \]
    \item \textbf{Port convention:} Port 1 corresponds to $Z_1$, and Port 2 corresponds to $Z_2$.
\end{itemize}

\subsubsection*{Reflection and Transmission Coefficients}
\begin{itemize}
    \item \textbf{Reflection at the junction from Port 2:}
    \[
    S_{22} = \Gamma_{A^+} = \frac{Z_1 - Z_2}{Z_1 + Z_2}
    \]
    \item \textbf{Voltage transmission coefficient:}
    \[
    \frac{V_A^-}{V_A^+} = \frac{2Z_1}{Z_1 + Z_2}
    \]
    \item \textbf{Power-normalized transmission:}
    \[
    S_{12} = \sqrt{\frac{Z_2}{Z_1}} \cdot \frac{V_A^-}{V_A^+}
    = \frac{2\sqrt{Z_1 Z_2}}{Z_1 + Z_2}
    \]
    \item \textbf{Symmetric result:} by reciprocity,
    \[
    S_{21} = \frac{2\sqrt{Z_1 Z_2}}{Z_1 + Z_2}
    \]
    \item \textbf{Reflection seen from Port 1:}
    \[
    S_{11} = \frac{Z_2 - Z_1}{Z_1 + Z_2}
    \]
\end{itemize}

\subsubsection*{Scattering Matrix}
\[
\mathbf{S} = \frac{1}{Z_1 + Z_2}
\begin{pmatrix}
Z_2 - Z_1 & 2\sqrt{Z_1 Z_2} \\
2\sqrt{Z_1 Z_2} & Z_1 - Z_2
\end{pmatrix}
\]

\subsubsection*{Device Properties}
\begin{itemize}
    \item \textbf{Reciprocal:} $S_{12} = S_{21}$.
    \item \textbf{Not symmetric:} generally $S_{11} \neq S_{22}$ unless $Z_1 = Z_2$.
    \item \textbf{Lossless junction:} the S-matrix is unitary, $\mathbf{S}\mathbf{S}^\dagger = \mathbf{I}$.
\end{itemize}


\subsection*{Fundamental Power Wave Definitions}
\begin{itemize}
    \item \textbf{Incident wave:} 
    \[
    a_i = \frac{V_i + Z_{ri} I_i}{2\sqrt{Z_{ri}}}
    \]
    \item \textbf{Reflected wave:}
    \[
    b_i = \frac{V_i - Z_{ri} I_i}{2\sqrt{Z_{ri}}}
    \]
    \item \textbf{Power normalization:} $|a_i|^2$ and $|b_i|^2$ represent power in watts.
    \item \textbf{Net real power at port $i$:}
    \[
    P_i = |a_i|^2 - |b_i|^2
    \]
    \item \textbf{Spatial propagation:}
    \[
    a(x) = a(0)e^{-jkx}, \qquad b(x) = b(0)e^{+jkx}
    \]
    \item \textbf{Reference impedance:} $Z_{ri}$ is the characteristic impedance used to normalize port waves.
\end{itemize}

\subsection*{General N-Port S-Matrix Framework}
\begin{itemize}
    \item \textbf{Black-box model:} internal circuit details are replaced by input-output relations between incident and reflected waves.
    \item \textbf{S-matrix relation:}
    \[
    \mathbf{b} = \mathbf{S}\mathbf{a}
    \]
    \item \textbf{Matrix size:} for an $N$-port network, $\mathbf{S}$ is an $N \times N$ matrix.
    \item \textbf{Scattering parameter definition:}
    \[
    S_{ij} = \left.\frac{b_i}{a_j}\right|_{a_k = 0 \ \text{for all}\ k \neq j}
    \]
    \item \textbf{Matched-load condition:} $a_k=0$ is realized by terminating all non-driven ports with their reference impedances.
\end{itemize}

\subsection*{Two-Port Network}
\begin{itemize}
    \item \textbf{Equations:}
    \[
    b_1 = S_{11}a_1 + S_{12}a_2, \qquad
    b_2 = S_{21}a_1 + S_{22}a_2
    \]
    \item \textbf{Matrix form:}
    \[
    \begin{bmatrix} b_1 \\ b_2 \end{bmatrix}
    =
    \begin{bmatrix}
    S_{11} & S_{12} \\
    S_{21} & S_{22}
    \end{bmatrix}
    \begin{bmatrix} a_1 \\ a_2 \end{bmatrix}
    \]
\end{itemize}


\subsection*{Directional Coupler Performance}
\begin{itemize}
    \item \textbf{Coupling factor (dB):}
    \[
    \text{Coupling (dB)} = -10\log\!\left(\frac{P_{\text{coupling forward}}}{P_{\text{incident}}}\right)
    \]
    \item \textbf{Isolation factor (dB):}
    \[
    \text{Isolation (dB)} = -10\log\!\left(\frac{P_{\text{coupled reverse}}}{P_{\text{incident}}}\right)
    \]
    \item \textbf{Directivity (dB):}
    \[
    \text{Directivity (dB)} = \text{Isolation (dB)} - \text{Coupling (dB)}
    \]
    \item \textbf{Mainline loss:} total insertion loss in the primary through-line.
\end{itemize}

\subsection*{S-Parameter Extractions}
\begin{itemize}
    \item \textbf{Forward measurements:} when SW1 is in position 1,
    \[
    S_{11} = \frac{b_1}{a_1}, \qquad S_{21} = \frac{b_2}{a_1}
    \]
    \item \textbf{Reverse measurements:} when SW1 is in position 2,
    \[
    S_{22} = \frac{b_2}{a_2}, \qquad S_{12} = \frac{b_1}{a_2}
    \]
    \item \textbf{2$\times$2 S-matrix:}
    \[
    \begin{bmatrix}
    S_{11} & S_{12} \\
    S_{21} & S_{22}
    \end{bmatrix}
    \]
    \item \textbf{Index convention:} $i$ = destination/output port, $j$ = source/input port.
\end{itemize}

\section{Signal Modulation}

\subsection*{General Sine Wave Equation}
\begin{itemize}
    \item \textbf{Time-domain representation:}
    \[
    v(t) = A \cdot \sin(2\pi ft + \phi)
    \]
    where $A$ is amplitude, $f$ is frequency, and $\phi$ is phase.
\end{itemize}

\subsection*{Amplitude Modulation (AM)}
\begin{itemize}
    \item \textbf{Time-domain equation:}
    \[
    s(t) = \left[A_c + A_m \cos 2\pi f_m t\right] \cos 2\pi f_c t
    \]
    where $A_c$ is carrier amplitude, $A_m$ is modulation amplitude, $f_c$ is carrier frequency, and $f_m$ is modulation frequency.
    \item \textbf{Trigonometric identity for frequency shifting:} 
    \[
    \cos A \cos B = \frac{1}{2}[\cos(A + B) + \cos(A - B)]
    \]
    \item \textbf{Expanded AM equation (sidebands):}
    \[
    s(t) = A_c \cos 2\pi f_c t + \frac{A_m}{2} \cos 2\pi(f_c + f_m)t + \frac{A_m}{2} \cos 2\pi(f_c - f_m)t
    \]
\end{itemize}

\subsection*{I/Q Modulation Time-Domain Equation}
\begin{itemize}
    \item \textbf{General form:}
    \[
    y(t) = g_1(t) \cos 2\pi f_c t - g_2(t) \sin 2\pi f_c t
    \]
    where $g_1(t)$ is the In-Phase (I) signal and $g_2(t)$ is the Quadrature (Q) signal.
\end{itemize}

\subsection*{Complex Baseband Signal Representation}
\begin{itemize}
    \item \textbf{Complex envelope:}
    \[
    g(t) = g_1(t) + j g_2(t)
    \]
    \item \textbf{Quadrature modulation (complex notation):}
    \[
    v(t) = \text{Re}\left\{g(t) e^{j2\pi f_c t}\right\}
    \]
    \item \textbf{Expanded form:}
    \[
    v(t) = g_1(t) \cos(2\pi f_c t) - g_2(t) \sin(2\pi f_c t)
    \]
\end{itemize}

\subsection*{Quadrature Demodulation Process}
\begin{itemize}
    \item \textbf{Total received signal ($x(t)$):}
    \[
    x(t) = y(t) e^{-j2\pi f_c t}
    \]
    \item \textbf{In-Phase extraction ($z_1(t)$):}
    \[
    z_1(t) = \frac{1}{2} g_1(t) + [\text{high-frequency terms}]
    \]
    \item \textbf{Quadrature extraction ($z_2(t)$):}
    \[
    z_2(t) = \frac{1}{2} g_2(t) + [\text{high-frequency terms}]
    \]
    (after low-pass filtering to remove high-frequency components)
\end{itemize}

\subsection*{Signal Examples}
\begin{itemize}
    \item \textbf{Sum of three tones (real):}
    \[
    s_1(t) = 10 \cos 2\pi 100t + \cos 2\pi 200t + 4 \cos 2\pi 300t
    \]
    \item \textbf{Sum of three tones with phase shift:}
    \[
    s_2(t) = 10 \cos\left(2\pi 100t + \frac{\pi}{4}\right) + \cos\left(2\pi 200t + \frac{\pi}{6}\right) + 4 \cos 2\pi 300t
    \]
\end{itemize}

\section{Digital Signal Processing}

\subsection*{Sampling and Aliasing}
\begin{itemize}
    \item \textbf{Sampling Rate ($f_s$):} The inverse of the sampling period ($t_s$).
    \[
    f_s = \frac{1}{t_s}
    \]
    \item \textbf{Nyquist-Shannon Criterion (Undersampling):} For a band-limited signal between frequencies $f_l$ and $f_h$.
    \[
    f_s > 2(f_h - f_l)
    \]
    \item \textbf{Alias Frequency:} The frequency at which a signal $f$ appears when sampled at $f_s$.
    \[
    f_{\text{alias}} = |f - n \cdot f_s|
    \]
    \item \textbf{Spectral Image Locations (DAC):} Positions of unwanted images after D/A conversion.
    \[
    \text{Location} = N \cdot f_s \pm f
    \]
    where $N$ is an integer and $f$ is the generated signal frequency.
\end{itemize}

\subsection*{Quantization and Performance Metrics}
\begin{itemize}
    \item \textbf{Quantization Step ($q$):} The voltage difference between two adjacent discrete levels.
    \[
    q = \frac{V_{\text{FS}}}{2^N - 1}
    \]
    where $V_{\text{FS}} = 2V_{\max}$ is the full-scale voltage.
    \item \textbf{ADC Resolution ($\Delta$):}
    \[
    \Delta = \frac{V_{\max} - V_{\min}}{2^n}
    \]
    \item \textbf{Dynamic Range (DR):} Logarithmic ratio between the largest and smallest measurable values.
    \[
    \text{DR} = 20 \log_{10}(2^N) \approx 6.02N \text{ [dB]}
    \]
    \item \textbf{Quantization Noise:} Idealized model of the noise characteristics.
    \[
    \text{Noise Power} = \frac{q^2}{12}
    \]
    \item \textbf{Spurious Free Dynamic Range (SFDR):}
    \[
    \text{SFDR} = \frac{P_{\text{fundamental}}}{P_{\text{biggest\_spur}}}
    \]
\end{itemize}

\subsection*{Fixed-Point Data Representation (Qn.b format)}
\begin{itemize}
    \item \textbf{Unsigned Range:}
    \begin{itemize}
        \item Minimum: $0$
        \item Maximum: $2^n - 2^{-b}$
    \end{itemize}
    \item \textbf{Signed 2's Complement Range:}
    \begin{itemize}
        \item Minimum: $-2^{n-1}$
        \item Maximum: $2^{n-1} - 2^{-b}$
    \end{itemize}
    \item \textbf{Precision/Resolution:} The smallest representable number.
    \[
    \text{Resolution} = 2^{-b}
    \]
    \item \textbf{Arithmetic Bit-Width Expansion:}
    \begin{itemize}
        \item Addition/Subtraction: Requires $n+1$ bits to avoid overflow.
        \item Multiplication: Multiplying $Q(m1.n1)$ and $Q(m2.n2)$ results in $Q(m1+m2).(n1+n2)$.
    \end{itemize}
\end{itemize}

\subsection*{Multirate Operations}
\begin{itemize}
    \item \textbf{Decimation:} Reducing the sampling rate by an integer factor $M$.
    \[
    f_d = \frac{f_s}{M}
    \]
    \item \textbf{Interpolation:} Increasing the sampling rate by an integer factor $L$.
    \[
    f_u = L \cdot f_s
    \]
    \item \textbf{Interpolated Image Location:} The location of the first image after interpolation by factor $K$.
    \[
    f_{\text{image}} = K \cdot f_c - f_o
    \]
\end{itemize}

\subsection*{Data Converter Hardware Dynamics}
\begin{itemize}
    \item \textbf{Interleaved ADC Sampling Phase:} The phase offset for the $n^{\text{th}}$ sub-ADC in a system of $m$ interleaved converters.
    \[
    \theta_n = \frac{2\pi(n - 1)}{m}
    \]
    \item \textbf{DAC Frequency Response (Normal Mode):} Follows a sinc envelope.
    \[
    H(f) = \frac{\sin(x)}{x}, \quad \text{where} \quad x = \frac{\pi f}{f_s}
    \]
    \item \textbf{DAC Frequency Response (Mix-Mode):} Reshapes energy for the 2nd Nyquist Zone.
    \[
    H(f) = \frac{\sin^2(x)}{x}, \quad \text{where} \quad x = \frac{\pi f}{2f_s}
    \]
\end{itemize}

\section{Semiconductor Qubits and the Elzerman Method}

\subsection*{Energy Level and Temperature Requirements}
\begin{itemize}
    \item \textbf{Zeeman Energy ($\Delta E_Z$):} Approximately $200 \, \mu\text{eV}$ at a magnetic field of 10 T.
    \item \textbf{Thermal Energy ($kT$):} Approximately $25 \, \mu\text{eV}$ at a temperature of 0.3 K.
    \item \textbf{Orbital Energy Spacing ($\Delta E_{\text{orbital}}$):} Approximately $1.1 \, \text{meV}$.
    \item \textbf{System Requirement Formula:} To ensure a valid qubit state, the parameters must satisfy:
    \[
    \Delta E_{\text{orbital}} \gg \Delta E_Z \gg kT
    \]
\end{itemize}

\subsection*{Tuning and Bias Formulas}
\begin{itemize}
    \item \textbf{Optimal Bias Point ($V_M$):} The voltage $V_M$ sets the bias point for energy level oscillations.
    \item \textbf{Optimal Working Point Calculation:} If the crossing point where the spin-up electron energy exceeds the Fermi level is $V_{M0}$, the optimal working point to place the Fermi energy halfway between levels is:
    \[
    V_M = V_{M0} + \frac{1}{2e} \Delta E_Z
    \]
\end{itemize}

\subsection*{Measurement Fidelity and Probability}
\begin{itemize}
    \item \textbf{Error Probability $\alpha$:} The probability of measuring ``down'' when the state is spin-up.
    \[
    \alpha = P(\text{``down''} \mid \text{spin-}\uparrow)
    \]
    \item \textbf{Error Probability $\beta$:} The probability of measuring ``up'' when the state is spin-down.
    \[
    \beta = P(\text{``up''} \mid \text{spin-}\downarrow)
    \]
    \item \textbf{Decay and Detection Rate ($\beta_1$):}
    \[
    \beta_1 = \frac{1}{1 + T_1 \Gamma_\downarrow}
    \]
    \item \textbf{Combined $\beta$ Estimation:}
    \[
    1 - \beta \sim (1 - \beta_1)(1 - \beta_2) + \alpha \beta_1
    \]
    \item \textbf{Spin-up Fidelity:} $1 - \alpha$
    \item \textbf{Visibility Formula:}
    \[
    \text{Visibility} = 1 - \alpha - \beta
    \]
\end{itemize}

\subsection*{Qubit Interaction and Control}
\begin{itemize}
    \item \textbf{Hamiltonian Basis:} The baseline Hamiltonian for the split levels is represented as:
    \[
    H_0 \sim \sum \omega_i \sigma_{zi}
    \]
    \item \textbf{Exchange Interaction ($J$):} In two-qubit systems, the exchange interaction is computed using the difference in resonance frequencies ($f$) of one qubit based on the state of the other:
    \[
    J = f_2 - f_1 \quad \text{or} \quad J = f_4 - f_3
    \]
    where $f_1/f_2$ are resonance frequencies for $q_0$ given $q_1$ states, and $f_3/f_4$ are for $q_1$ given $q_0$ states.
\end{itemize}

\subsection*{Randomized Benchmarking (RB)}
\begin{itemize}
    \item \textbf{Data Fitting:} Success probabilities $p(m)$ for a sequence of $m$ gates are processed using:
    \[
    p(m) \text{ fitted to a linear combination of exponential decays}
    \]
    \item \textbf{SPAM Fidelity Estimation:} State Preparation and Measurement (SPAM) error can be estimated by evaluating the least-squares fit of the RB decay curve at:
    \[
    m = 0
    \]
\end{itemize}
